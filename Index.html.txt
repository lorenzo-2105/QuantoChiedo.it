<script>
    function toggleTravelInputs() {
        const mode = document.getElementById('travelPayer').value;
        const fields = document.querySelectorAll('.travel-field');
        
        fields.forEach(f => {
            if (mode === 'my_charge') {
                f.style.display = 'block';
            } else {
                f.style.display = 'none';
            }
        });
    }

    function calculate() {
        const netTarget = parseFloat(document.getElementById('netTarget').value) || 0;
        const hours = parseFloat(document.getElementById('hours').value) || 1;
        const expenses = parseFloat(document.getElementById('expenses').value) || 0;
        const rivalsaPct = parseFloat(document.getElementById('rivalsa').value) || 0;
        const regime = document.getElementById('regime').value;

        // Dati Trasferta
        const travelPayer = document.getElementById('travelPayer').value;
        const kmDist = parseFloat(document.getElementById('kmDist').value) || 0;
        const travelCost = parseFloat(document.getElementById('travelCost').value) || 0;
        const hotelCost = parseFloat(document.getElementById('hotelCost').value) || 0;

        if (netTarget <= 0) {
            alert('Inserisci un importo netto valido');
            return;
        }

        let totalTravel = 0;
        if (travelPayer === 'my_charge') {
            totalTravel = travelCost + hotelCost;
        }

        // Il compenso base comprende il netto + spese vive/materiali + rimborso trasferte
        const baseTarget = netTarget + expenses + totalTravel;

        let grossBeforeTax = baseTarget;
        let rivalsaVal = 0;

        if (rivalsaPct > 0) {
            rivalsaVal = baseTarget * (rivalsaPct / 100);
            grossBeforeTax += rivalsaVal;
        }

        let totalGross = 0;
        let tax = 0;

        if (regime === 'prestazione') {
            totalGross = grossBeforeTax / 0.80;
            tax = totalGross * 0.20;
        } else if (regime === 'forfettario') {
            totalGross = grossBeforeTax / 0.85;
            tax = totalGross * 0.15;
        } else {
            totalGross = grossBeforeTax;
            tax = 0;
        }

        const hourlyRate = netTarget / hours;

        document.getElementById('resNet').innerText = '€ ' + netTarget.toFixed(2);
        document.getElementById('resExpenses').innerText = '€ ' + expenses.toFixed(2);
        document.getElementById('resTax').innerText = '€ ' + tax.toFixed(2);
        document.getElementById('resHourly').innerText = '€ ' + hourlyRate.toFixed(2) + ' / ora';
        document.getElementById('resGross').innerText = '€ ' + totalGross.toFixed(2);

        if (totalTravel > 0) {
            document.getElementById('resTravel').innerText = '€ ' + totalTravel.toFixed(2);
            document.getElementById('rowTravel').style.display = 'flex';
        } else {
            document.getElementById('rowTravel').style.display = 'none';
        }

        if (rivalsaVal > 0) {
            document.getElementById('resRivalsa').innerText = '€ ' + rivalsaVal.toFixed(2);
            document.getElementById('rowRivalsa').style.display = 'flex';
        } else {
            document.getElementById('rowRivalsa').style.display = 'none';
        }

        // Generazione Testo Email
        let msg = `Gentile Cliente,

in merito alla richiesta, le confermo il preventivo dettagliato per lo svolgimento delle attività (stima di ${hours} ore lavorative):
`;
        
        const feeAlone = totalGross - expenses - totalTravel;
        msg += `
- Compenso professionale: € ${feeAlone.toFixed(2)}`;
        
        if (expenses > 0) {
            msg += `
- Rimborso acquisti materiali/licenze extra: € ${expenses.toFixed(2)}`;
        }

        if (travelPayer === 'my_charge' && totalTravel > 0) {
            msg += `
- Rimborso spese di trasferta e pernottamento`;
            if (kmDist > 0) msg += ` (${kmDist} km A/R)`;
            msg += `: € ${totalTravel.toFixed(2)}`;
        } else if (travelPayer === 'client_charge') {
            msg += `
- Note trasferta: Spese di viaggio e alloggio intese a carico diretto del Committente.`;
        }

        msg += `
---------------------------------`;
        msg += `
TOTALE LORDO: € ${totalGross.toFixed(2)}`;

        if (regime === 'prestazione') {
            msg += `

*Nota fiscale:* L'importo è soggetto a ritenuta d'acconto del 20% (€ ${tax.toFixed(2)}), che provvederete a versare per mio conto. L'importo netto corrisposto sarà di € ${(totalGross - tax).toFixed(2)}.`;
        }

        msg += `

Resto a disposizione per qualsiasi chiarimento e per confermare la data di inizio lavori.`;

        document.getElementById('emailText').innerText = msg;
        document.getElementById('resultsBlock').style.display = 'block';
    }
</script>
