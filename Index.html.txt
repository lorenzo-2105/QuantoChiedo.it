<script>
    function calculate() {
        const netTarget = parseFloat(document.getElementById('netTarget').value);
        const regime = document.getElementById('regime').value;
        
        if (isNaN(netTarget) || netTarget <= 0) {
            alert('Inserisci un importo valido');
            return;
        }

        let gross = 0;
        let tax = 0;

        if (regime === 'prestazione') {
            // Formula ritenuta d'acconto 20% sul lordo: Netto = Lordo * 0.80 -> Lordo = Netto / 0.80
            gross = netTarget / 0.80;
            tax = gross * 0.20;
        } else if (regime === 'forfettario') {
            // Stima approssimativa forfettario al 15%
            gross = netTarget / 0.85;
            tax = gross * 0.15;
        } else {
            gross = netTarget;
            tax = 0;
        }

        document.getElementById('resNet').innerText = '€ ' + netTarget.toFixed(2);
        document.getElementById('resTax').innerText = '€ ' + tax.toFixed(2);
        document.getElementById('resGross').innerText = '€ ' + gross.toFixed(2);

        // Generazione messaggio automatico
        let msg = `Buongiorno,\nin merito al lavoro richiesto, il preventivo complessivo è di € ${gross.toFixed(2)} lordi.`;
        
        if (regime === 'prestazione') {
            msg += `\n\nTrattandosi di prestazione occasionale, verrà applicata la ritenuta d'acconto del 20% (€ ${tax.toFixed(2)}), che verserete voi per mio conto. L'importo netto da me percepito sarà di € ${netTarget.toFixed(2)}.`;
        }

        msg += `\n\nResto a disposizione per qualsiasi chiarimento.`;

        document.getElementById('emailText').innerText = msg;
        document.getElementById('resultsBlock').style.display = 'block';
    }
</script>
